cmake_minimum_required(VERSION 3.20)
project(JoltSoftBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(USE_STATIC_MSVC_RUNTIME_LIBRARY OFF CACHE BOOL "" FORCE)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(USE_ASSERTS OFF CACHE BOOL "" FORCE)
set(FLOATING_POINT_EXCEPTIONS_ENABLED OFF CACHE BOOL "" FORCE)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(ENABLE_ALL_WARNINGS OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(JPH_USE_DX12 OFF CACHE BOOL "" FORCE)
set(JPH_USE_VK OFF CACHE BOOL "" FORCE)
set(JPH_USE_MTL OFF CACHE BOOL "" FORCE)
# Compare the SSE2 CPU baseline first. A separate AVX2 build can follow.
foreach(feature SSE4_1 SSE4_2 AVX AVX2 AVX512 LZCNT TZCNT F16C FMADD)
  set(USE_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../reference/JoltPhysics/Build" jolt)
add_executable(JoltSoftBenchmark main.cpp)
target_link_libraries(JoltSoftBenchmark PRIVATE Jolt)
if(MSVC)
  target_compile_options(JoltSoftBenchmark PRIVATE /O2 /fp:fast /W4)
endif()
